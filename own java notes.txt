import java.util.Scanner;

public class ReverseofNum {
    public static void main(String args[]) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        
        // Just call the method; it will handle the printing itself
        meth(n); 
    }

    // 'void' means this method performs an action but returns nothing
    public static void meth(int n) {
        int rev = 0;
        int rem;
        while (n != 0) {
            rem = n % 10;
            rev = rev * 10 + rem;
            n = n / 10;
        }
        // Because it is void, we MUST print the result right here
        System.out.println(rev); 
    }
}




import java.util.Scanner;

public class ReverseofNum {
    public static void main(String args[]) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        
        // 'main' receives the number back and prints it directly
        System.out.println(meth(n)); 
    }

    // 'int' means this method MUST give back a whole number
    public static int meth(int n) {
        int rev = 0;
        int rem;
        while (n != 0) {
            rem = n % 10;
            rev = rev * 10 + rem;
            n = n / 10;
        }
        // Pass the final value back to whoever called this method
        return rev; 
    }
}
